library(dplyr)
library(data.table)
library(magrittr)
library(ggplot2)
library(readxl)
library(clusterProfiler)
library(enrichplot)
yap = read_excel("data/P5vsP1/41586_2015_BFnature15382_MOESM27_ESM.xlsx")%>%
  as.data.frame()%>%
  dplyr::select(Gene_Symbol, Fold_log2)%>%
  dplyr::filter(Fold_log2 < -2)

de_3dveh = readRDS("data/KrasInhibition/inhibit3d_veh_res.rds")
de_14dveh = readRDS("data/KrasInhibition/inhibit14d_veh_res.rds")



feature_df = data.frame(set = rep("YAP", length(yap$Gene_Symbol)),
                        gene = yap$Gene_Symbol,
                        stringsAsFactors = F)

3 day vs vehicle

res_df_3_av = de_3dveh%>%
  dplyr::select(symbol, t)%>%
  dplyr::group_by(symbol)%>%
  dplyr::summarise(t = mean(t))

geneList = res_df_3_av$t
names(geneList) = res_df_3_av$symbol
geneList = sort(geneList, decreasing = T)



set.seed(47382686)
y <- GSEA(geneList, TERM2GENE = feature_df, pvalueCutoff = 1, minGSSize = 5, maxGSSize = 700)
## preparing geneSet collections...
## GSEA analysis...
## Warning in fgseaMultilevel(pathways = pathways, stats = stats, minSize =
## minSize, : For some pathways, in reality P-values are less than 1e-10. You can

## leading edge analysis...
## done...
nes = paste(paste0(y@result$Description, ":", round(y@result$NES,2)), collapse = "; ")

gseaplot2(y, geneSetID = 1,pvalue_table = T, color = "darkorange",
           title = paste0("YAP 3d NES ",nes))
