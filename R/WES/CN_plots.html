boxplot stratifying log2 fc according to CN change between P5 and P1

res.expr = readRDS("data/P5vsP1/res_P5_P1.rds")

res.cn = readRDS("data/WES/res.cn.rds")%>%
  dplyr::mutate(p5_p1_change = m484.liver.p5 - m4.colon.p1)%>%
  dplyr::select(Symbol, p5_p1_change)%>%
  dplyr::group_by(Symbol)%>%
  dplyr::summarise(cnChange = mean(p5_p1_change, na.rm =T))%>%
  dplyr::ungroup()%>%
  as.data.frame()


expr_cn = res.expr%>%
  dplyr::select(symbol, LogFC)%>%
  dplyr::group_by(symbol)%>%
  dplyr::summarise(exprChange = mean(LogFC, na.rm =T))%>%
  dplyr::ungroup()%>%
  as.data.frame()%>%
  dplyr::left_join(res.cn, by = c("symbol" = "Symbol"))%>%
  dplyr::mutate(cnChange = as.factor(cnChange))%>%
  na.omit()%>%
  dplyr::filter(cnChange != -3) ### an outlier 
p = ggplot(data = expr_cn, aes(cnChange, exprChange)) +
  geom_violin(alpha = 0.5, aes(color = cnChange), fill = "grey") +
  geom_quasirandom(size = 0.01, pch = 16, aes(color = cnChange)) +
  geom_boxplot(width = 0.1, outlier.shape = NA, alpha = 0.5) +
  theme_bw() +
  labs(y = "exprChange (log2 fold)", title = "comparing P5 to P1")



result <- JonckheereTerpstraTest(exprChange ~ cnChange, data = expr_cn, alternative = "increasing")
